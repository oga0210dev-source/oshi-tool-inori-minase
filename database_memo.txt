/////////////////////////////////
マスターテーブル
/////////////////////////////////
create table public.m_user (
  user_id text not null,
  user_name text not null,
  password text not null,
  role text not null default 'user'::text,
  profile_image text null,
  member_since date null,
  email text null,
  gender text null,
  birthday date null,
  prefecture smallint null,
  x_account text null,
  instagram_account text null,
  discord_account text null,
  profile_message text null,
  is_active boolean not null default true,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  constraint m_user_pkey1 primary key (user_id),
  constraint m_user_prefecture_fkey foreign KEY (prefecture) references m_prefecture (prefecture_code)
) TABLESPACE pg_default;

create trigger update_m_user_updated_at BEFORE
update on m_user for EACH row
execute FUNCTION update_updated_at_column ();

create table public.m_user_public_setting (
  user_id character varying(50) not null,
  gender_public smallint not null default 0,
  birthday_public smallint not null default 0,
  age_public smallint not null default 0,
  member_since_public smallint not null default '0'::smallint,
  prefecture_public smallint not null default '0'::smallint,
  sns_public smallint not null default '0'::smallint,
  live_public smallint not null default 0,
  meeting_public smallint not null default 0,
  created_at timestamp without time zone null default CURRENT_TIMESTAMP,
  updated_at timestamp without time zone null default CURRENT_TIMESTAMP,
  constraint m_user_public_setting_pkey primary key (user_id),
  constraint m_user_public_setting_user_id_fkey foreign KEY (user_id) references m_user (user_id) on update CASCADE on delete CASCADE
) TABLESPACE pg_default;

create trigger update_m_user_public_setting_updated_at BEFORE
update on m_user_public_setting for EACH row
execute FUNCTION update_updated_at_column ();

create table public.m_prefecture (
  prefecture_code smallint not null,
  prefecture_name character varying(20) not null,
  area_name character varying(20) null,
  display_order smallint not null,
  is_overseas boolean not null default false,
  is_active boolean not null default true,
  constraint m_prefecture_pkey primary key (prefecture_code)
) TABLESPACE pg_default;

create table public.m_song (
  song_id bigserial not null,
  song_group_id bigserial not null,
  song_name character varying(100) not null,
  release_date date null,
  album_name character varying(100) null,
  display_order integer null,
  lyricist character varying(100) null,
  composer character varying(100) null,
  arranger character varying(100) null,
  tie_up character varying(200) null,
  youtube_url text null,
  apple_music_url text null,
  spotify_url text null,
  is_public boolean not null default true,
  is_deleted boolean null default false,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  constraint m_song_pkey primary key (song_id),
  constraint unique_song_album unique (album_name, song_name)
) TABLESPACE pg_default;

create trigger update_m_song_updated_at BEFORE
update on m_song for EACH row
execute FUNCTION update_updated_at_column ();

create table public.m_live (
  live_id bigserial not null,
  live_name character varying(200) not null,
  tour_name character varying(200) null,
  tour_order integer null,
  live_date date not null,
  venue_name character varying(200) not null,
  prefecture_code smallint null,
  blu_ray_url text null,
  official_url text null,
  public_flag boolean not null default true,
  is_deleted boolean null default false,
  created_at timestamp without time zone null default CURRENT_TIMESTAMP,
  updated_at timestamp without time zone null default CURRENT_TIMESTAMP,
  constraint m_live_pkey primary key (live_id),
  constraint fk_live_prefecture foreign KEY (prefecture_code) references m_prefecture (prefecture_code)
) TABLESPACE pg_default;

create trigger update_m_live_updated_at BEFORE
update on m_live for EACH row
execute FUNCTION update_updated_at_column ();

create table public.m_setlist (
  event_type character varying(20) not null,
  event_id bigint not null,
  song_id bigint not null,
  song_order integer not null,
  is_medley boolean not null default false,
  medley_order integer null,
  created_at timestamp without time zone null default CURRENT_TIMESTAMP,
  updated_at timestamp without time zone null default CURRENT_TIMESTAMP,
  constraint m_setlist_pkey primary key (event_type, event_id, song_id),
  constraint fk_setlist_song foreign KEY (song_id) references m_song (song_id),
  constraint chk_event_type check (
    (
      (event_type)::text = any (
        array[
          ('LIVE'::character varying)::text,
          ('CHOMIN'::character varying)::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create unique INDEX IF not exists idx_m_setlist_order on public.m_setlist using btree (event_type, event_id, song_order) TABLESPACE pg_default;

create trigger update_m_setlist_updated_at BEFORE
update on m_setlist for EACH row
execute FUNCTION update_updated_at_column ();










/////////////////////////////////
ライブデータ　トランザクション
/////////////////////////////////
create table public.t_live_user (
  user_id text not null,
  live_id bigint not null,
  is_join boolean not null default true,
  created_at timestamp without time zone not null default now(),
  updated_at timestamp without time zone not null default now(),
  constraint pk_t_live_user primary key (user_id, live_id),
  constraint fk_t_live_user_live foreign KEY (live_id) references m_live (live_id),
  constraint fk_t_live_user_user foreign KEY (user_id) references m_user (user_id)
) TABLESPACE pg_default;

create trigger update_t_live_user_updated_at BEFORE
update on t_live_user for EACH row
execute FUNCTION update_updated_at_column ();
