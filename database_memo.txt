create table public.m_maintenance (
  created_at timestamp without time zone not null default CURRENT_TIMESTAMP,
  updated_at timestamp without time zone not null default CURRENT_TIMESTAMP,
  maintenance_id bigserial not null,
  announcement_id bigint null,
  maintenance_type character varying(20) not null,
  target_key character varying(100) null,
  title character varying(200) not null,
  message text null,
  start_at timestamp without time zone null,
  end_at timestamp without time zone null,
  is_emergency boolean not null default false,
  is_active boolean not null default false,
  constraint m_maintenance_pkey primary key (maintenance_id),
  constraint fk_m_maintenance_announcement foreign KEY (announcement_id) references m_announcement (announcement_id),
  constraint chk_maintenance_partial_target check (
    (
      (
        ((maintenance_type)::text = 'ALL'::text)
        and (target_key is null)
      )
      or (
        ((maintenance_type)::text = 'PARTIAL'::text)
        and (target_key is not null)
      )
    )
  ),
  constraint chk_maintenance_type check (
    (
      (maintenance_type)::text = any (
        (
          array[
            'ALL'::character varying,
            'PARTIAL'::character varying
          ]
        )::text[]
      )
    )
  )
) TABLESPACE pg_default;

  inquiry_id bigserial not null,
  user_id text not null,
  inquiry_type character varying(20) not null,
  subject character varying(100) not null,
  email character varying(255) null,
  message text not null,
  status character varying(20) not null default 'UNRESOLVED'::character varying,
  admin_memo text null,
  created_at timestamp without time zone not null default CURRENT_TIMESTAMP,
  updated_at timestamp without time zone not null default CURRENT_TIMESTAMP,
  constraint t_inquiry_pkey primary key (inquiry_id),
  constraint fk_t_inquiry_user foreign KEY (user_id) references m_user (user_id) on delete CASCADE,
  constraint chk_t_inquiry_email check (
    (
      (
        ((inquiry_type)::text = 'INQUIRY'::text)
        and (email is not null)
        and ((email)::text <> ''::text)
      )
      or ((inquiry_type)::text <> 'INQUIRY'::text)
    )
  ),
  constraint chk_t_inquiry_status check (
    (
      (status)::text = any (
        (
          array[
            'UNRESOLVED'::character varying,
            'IN_PROGRESS'::character varying,
            'RESOLVED'::character varying
          ]
        )::text[]
      )
    )
  ),
  constraint chk_t_inquiry_type check (
    (
      (inquiry_type)::text = any (
        (
          array[
            'INQUIRY'::character varying,
            'REQUEST'::character varying,
            'BUG'::character varying,
            'TODO'::character varying
          ]
        )::text[]
      )
    )
  )
) TABLESPACE pg_default;
